import { alertClient } from "./alertClient";
import {
  ListAlertsQuery,
  ListAlertsResponse,
  AlertItem,
  AcknowledgeAlertRequest,
} from "../types/alert";

export const alertApi = {
  listAlerts: async (
    params?: ListAlertsQuery
  ): Promise<ListAlertsResponse> => {
    const response = await alertClient.get<ListAlertsResponse>(
      "/api/v1/alerts",
      {
        params,
      }
    );

    if (!params?.page || params.page === 1) {
      console.log("[AlertAPI] listAlerts response shape:", {
        rowsCount:
          response.data?.rows?.length ??
          "rows field missing",
        total: response.data?.total,
        hasMore: response.data?.hasMore,
        hasTotals: Boolean(response.data?.totals),
        firstAlertKeys: response.data?.rows?.[0]
          ? Object.keys(response.data.rows[0])
          : "no alerts",
      });
    }

    return response.data;
  },

  getAlertById: async (
    id: string
  ): Promise<AlertItem> => {
    const response = await alertClient.get<{
      data: AlertItem;
    }>(`/api/v1/alerts/${id}`);

    return response.data.data;
  },

  acknowledgeAlert: async (
    id: string,
    data?: AcknowledgeAlertRequest
  ): Promise<{
    message: string;
    data?: AlertItem;
  }> => {
    const response = await alertClient.patch<{
      message: string;
      data?: AlertItem;
    }>(
      `/api/v1/alerts/${id}/ack`,
      data || { action: "ack" }
    );

    return response.data;
  },

  unacknowledgeAlert: async (
    id: string
  ): Promise<{
    message: string;
    data?: AlertItem;
  }> => {
    const response = await alertClient.patch<{
      message: string;
      data?: AlertItem;
    }>(
      `/api/v1/alerts/${id}/ack`,
      { action: "unack" }
    );

    return response.data;
  },

  bulkAcknowledge: async (
    params: {
      action: "bulk_ack" | "bulk_unack";
      preset:
        | "today"
        | "yesterday"
        | "one_week"
        | "one_month"
        | "before_today"
        | "before_yesterday"
        | "before_one_week"
        | "before_one_month";
      did?: string;
    }
  ): Promise<{
    message: string;
    matched?: number;
    modified?: number;
  }> => {
    const response = await alertClient.patch<{
      message: string;
      matched?: number;
      modified?: number;
    }>("/api/v1/alerts/bulk_ack", params);

    return response.data;
  },

  /**
   * IMPORTANT:
   * This is the legacy CLEAR operation.
   *
   * DO NOT use this for "Mark as Solved".
   * clear => is_cleared=true => alert disappears
   * from normal Alert API lists.
   */
  clearAlert: async (
    params: {
      action: "clear" | "unclear";
      preset:
        | "today"
        | "yesterday"
        | "this_week"
        | "last_week"
        | "this_month"
        | "last_month"
        | "custom"
        | "all_for_did";
      note?: string;
      start?: string;
      end?: string;
      did?: string;
      severity?: string;
    }
  ): Promise<{
    message: string;
    matched?: number;
    modified?: number;
  }> => {
    const response = await alertClient.patch<{
      message: string;
      matched?: number;
      modified?: number;
    }>("/api/v1/alerts/clear", params);

    return response.data;
  },

  getBuildMeta: async (): Promise<{
    message: string;
    company_id: string;
  }> => {
    const response = await alertClient.get<{
      message: string;
      company_id: string;
    }>("/api/v1/alerts/build");

    return response.data;
  },
};

/**
 * alertClient.ts
 * ----------------------------------------------------------------------------
 * Dedicated Axios instance for the deployed ALERT backend.
 *
 * Production base URL: https://prod.esmbackend.click/api/alert
 *
 * Design principles:
 *  - DEFAULT is always the deployed HTTPS URL â€” never a local address
 *  - Stale local/HTTP URLs in AsyncStorage are REJECTED at read-time (not
 *    in a separate async migration that can race with the first request)
 *  - Dev Settings (gear icon on login screen) can override to a local backend
 *    but ONLY stores the override explicitly this session
 *  - Every request logs exact baseURL + auth state (no token values)
 *  - 401 auto-refresh via auth backend with queue for concurrent requests
 * ----------------------------------------------------------------------------
 */

import axios, { AxiosError, InternalAxiosRequestConfig } from "axios";
import { storage } from "../services/storage";
import { CONFIG } from "../config/constants";
import { authClient } from "./authClient";
import { RefreshTokenResponse } from "../types/auth";

// ============================================================================
// URL helpers
// ============================================================================

/**
 * Returns true if a URL is a local/development address that only works on an
 * emulator or local machine â€” NOT on a physical device over the internet.
 *
 * NOTE: http://192.168.x.x addresses are allowed because the user may be
 * deliberately testing against a LAN server via Dev Settings.
 */
const isStaleDevelopmentUrl = (url: string): boolean => {
  if (!url) return false;
  return (
    url.includes("10.0.2.2") ||   // Android emulator host alias
    url.includes("localhost") ||   // iOS simulator / local
    url.includes("127.0.0.1") ||   // loopback
    (url.includes("esmbackend.click") && !url.includes("/api/alert")) // stale path missing /api/alert
  );
};

// ============================================================================
// Runtime URL management
// ============================================================================

/** In-memory URL override. null = use CONFIG.DEFAULT_ALERT_BASE_URL. */
let _alertBaseUrl: string | null = null;

/**
 * Returns the effective alert backend base URL.
 *
 * Priority:
 *   1. In-memory _alertBaseUrl (set by setAlertBaseUrl / resetAlertBaseUrl)
 *   2. AsyncStorage stored value â€” ONLY if it is NOT a stale dev URL
 *   3. CONFIG.DEFAULT_ALERT_BASE_URL (deployed production HTTPS)
 *
 * Stale dev URLs (10.0.2.2, localhost, 127.0.0.1) are silently rejected and
 * cleared from storage. This prevents "Network Error" on physical devices
 * that still have an old emulator URL saved from a previous dev session.
 */
export const getAlertBaseUrl = async (): Promise<string> => {
  // Fast path: in-memory override already resolved this session
  if (_alertBaseUrl !== null) return _alertBaseUrl;

  try {
    const stored = await storage.getItem(CONFIG.STORAGE_KEYS.ALERT_BASE_URL);

    if (stored) {
      if (isStaleDevelopmentUrl(stored)) {
        // Stale emulator/localhost URL â€” clear it and fall through to deployed default
        console.warn(
          "[AlertClient] Stale dev URL in storage:",
          stored,
          "â†’ clearing, reverting to deployed URL"
        );
        storage.removeItem(CONFIG.STORAGE_KEYS.ALERT_BASE_URL).catch(() => {});
        // Fall through to default below
      } else {
        // Valid stored override (e.g. production URL or user-set LAN URL)
        _alertBaseUrl = stored;
        return stored;
      }
    }
  } catch (e) {
    console.warn("[AlertClient] getAlertBaseUrl storage read error:", e);
  }

  // Default: deployed production URL
  _alertBaseUrl = CONFIG.DEFAULT_ALERT_BASE_URL;
  return _alertBaseUrl;
};

/** Sets alert backend URL (Dev Settings only). Clears in-memory cache. */
export const setAlertBaseUrl = async (url: string): Promise<void> => {
  const formatted = url.trim().replace(/\/+$/, "");
  _alertBaseUrl = formatted;
  await storage.setItem(CONFIG.STORAGE_KEYS.ALERT_BASE_URL, formatted);
  alertClient.defaults.baseURL = formatted;
  console.log("[AlertClient] URL overridden to:", formatted);
};

/** Resets any Dev Settings URL override â†’ back to deployed production URL. */
export const resetAlertBaseUrl = async (): Promise<void> => {
  _alertBaseUrl = CONFIG.DEFAULT_ALERT_BASE_URL;
  await storage.removeItem(CONFIG.STORAGE_KEYS.ALERT_BASE_URL);
  alertClient.defaults.baseURL = CONFIG.DEFAULT_ALERT_BASE_URL;
  console.log("[AlertClient] URL reset to deployed:", CONFIG.DEFAULT_ALERT_BASE_URL);
};

// ============================================================================
// Request interceptor: resolve URL + inject Bearer token
// ============================================================================
alertClient.interceptors.request.use(
  async (config: InternalAxiosRequestConfig) => {
    // Resolve effective URL (validates + clears stale dev URLs inline)
    let baseUrl = await getAlertBaseUrl();
    baseUrl = baseUrl.trim().replace(/\/+$/, "");

    // Cleanly normalize URL path:
    // If baseUrl already ends with /api/v1 and config.url starts with /api/v1,
    // strip the duplicate /api/v1 from config.url.
    if (baseUrl.endsWith("/api/v1") && config.url?.startsWith("/api/v1")) {
      config.url = config.url.replace(/^\/api\/v1/, "");
    }
    if (config.url && !config.url.startsWith("/")) {
      config.url = `/${config.url}`;
    }

    config.baseURL = baseUrl;

    // Inject Bearer token
    const token = await storage.getItem(CONFIG.STORAGE_KEYS.ACCESS_TOKEN);
    if (token && !config.headers.Authorization) {
      config.headers.Authorization = `Bearer ${token}`;
    }

    // Safe diagnostic log â€” no token values
    console.log("[AlertClient] â–¶ Request:", config.method?.toUpperCase(), {
      baseURL: config.baseURL,
      path: config.url,
      fullURL: `${config.baseURL}${config.url}`,
      hasAuth: Boolean(config.headers.Authorization),
      tokenLen: token?.length ?? 0,
    });

    return config;
  },
  (err) => {
    console.error("[AlertClient] Request interceptor error:", err);
    return Promise.reject(err);
  }
);

// ============================================================================
// Response interceptor: error logging + 401 token refresh
// ============================================================================
let isRefreshing = false;
let failedQueue: Array<{
  resolve: (value?: unknown) => void;
  reject: (reason?: unknown) => void;
}> = [];

const processQueue = (err: unknown, token: string | null = null) => {
  failedQueue.forEach((p) => (err ? p.reject(err) : p.resolve(token)));
  failedQueue = [];
};

type UnauthorizedHandler = () => void;
let _onUnauthorizedHandler: UnauthorizedHandler | null = null;

export const setOnUnauthorizedHandler = (handler: UnauthorizedHandler | null) => {
  _onUnauthorizedHandler = handler;
};

alertClient.interceptors.response.use(
  (res) => {
    console.log("[AlertClient] â—€ Response:", res.status, res.config.url, {
      dataKeys: res.data ? Object.keys(res.data) : [],
    });
    return res;
  },
  async (error: AxiosError) => {
    // Full safe error log
    console.log("[AlertClient] âœ— Error:", {
      message: error?.message,
      code: error?.code,
      status: error?.response?.status,
      responseData: error?.response?.data,
      baseURL: error?.config?.baseURL,
      url: error?.config?.url,
      fullURL: `${error?.config?.baseURL ?? ""}${error?.config?.url ?? ""}`,
      method: error?.config?.method,
      hasAuthHeader: Boolean(error?.config?.headers?.Authorization),
      isNetworkError: !error?.response,
    });

    const originalRequest = error.config as InternalAxiosRequestConfig & {
      _retry?: boolean;
    };

    if (!error.response || !originalRequest) {
      return Promise.reject(error);
    }

    if (error.response.status === 401 && !originalRequest._retry) {
      if (isRefreshing) {
        return new Promise((resolve, reject) =>
          failedQueue.push({ resolve, reject })
        )
          .then((tkn) => {
            originalRequest.headers.Authorization = `Bearer ${tkn}`;
            return alertClient(originalRequest);
          })
          .catch((e) => Promise.reject(e));
      }

      originalRequest._retry = true;
      isRefreshing = true;

      try {
        const refreshToken = await storage.getItem(
          CONFIG.STORAGE_KEYS.REFRESH_TOKEN
        );
        if (!refreshToken) throw new Error("No refresh token in storage");

        const res = await authClient.post<RefreshTokenResponse>(
          "/api/v1/auth/refresh-token",
          { token: refreshToken }
        );

        const {
          accessToken,
          refreshToken: newRefreshToken,
          userData,
        } = res.data;

        if (!accessToken)
          throw new Error("Refresh response missing accessToken");

        await storage.setItem(CONFIG.STORAGE_KEYS.ACCESS_TOKEN, accessToken);
        if (newRefreshToken)
          await storage.setItem(
            CONFIG.STORAGE_KEYS.REFRESH_TOKEN,
            newRefreshToken
          );
        if (userData)
          await storage.setObject(CONFIG.STORAGE_KEYS.USER_DATA, userData);

        alertClient.defaults.headers.common.Authorization = `Bearer ${accessToken}`;
        originalRequest.headers.Authorization = `Bearer ${accessToken}`;

        processQueue(null, accessToken);
        return alertClient(originalRequest);
      } catch (refreshErr) {
        processQueue(refreshErr, null);
        await storage.removeItem(CONFIG.STORAGE_KEYS.ACCESS_TOKEN);
        await storage.removeItem(CONFIG.STORAGE_KEYS.REFRESH_TOKEN);
        await storage.removeItem(CONFIG.STORAGE_KEYS.USER_DATA);
        if (_onUnauthorizedHandler) {
          try {
            _onUnauthorizedHandler();
          } catch (e) {
            console.warn("[AlertClient] onUnauthorizedHandler error:", e);
          }
        }
        return Promise.reject(refreshErr);
      } finally {
        isRefreshing = false;
      }
    }

    return Promise.reject(error);
  }
);

/**
 * AlertCard.tsx â€” PowerIQ Alert Item Card
 * ----------------------------------------------------------------------------
 * Industrial alert card matching reference layout:
 *  - Clean white surface with subtle border
 *  - Severity badge (Critical, Warning, Info)
 *  - Acknowledged badge only when backend status is acknowledged
 *  - Relative timestamp with clock icon + menu dots
 *  - Panel Name
 *  - PERFORMANCE METRIC (human-readable website name, never raw tags)
 *  - Recorded Value (max 3 decimals, no "pf" for Power Factor)
 *  - Alert Message
 *  - Exact date/time with calendar icon
 *  - Action area:
 *      * Unacknowledged: "Acknowledge" button (text only, NO check icon)
 *      * Acknowledged (not solved): "Mark as Solved" button
 *      * Solved: No action button
 *  - "Details >" button
 * ----------------------------------------------------------------------------
 */

import React from "react";
import {
  View,
  Text,
  StyleSheet,
  TouchableOpacity,
} from "react-native";
import { AlertItem, AlertSeverity, isAlertAcknowledged, isAlertSolved } from "../types/alert";
import { formatDistanceToNow } from "date-fns";
import { formatAlertCardDate } from "../utils/dateFormatter";
import {
  AlertCircle,
  AlertTriangle,
  Info,
  CheckCircle2,
  Clock,
  ChevronRight,
  Calendar,
  MoreVertical,
} from "lucide-react-native";
import {
  getPerformanceMetricName,
  formatAlertValue,
  formatAlertUnit,
} from "../utils/metrics";

interface Props {
  alert: AlertItem;
  onPress: () => void;
  onAcknowledgePress: (alert: AlertItem) => void;
  onMarkAsSolvedPress: (alert: AlertItem) => void;
}

export const AlertCard: React.FC<Props> = ({
  alert,
  onPress,
  onAcknowledgePress,
  onMarkAsSolvedPress,
}) => {
  const severity: AlertSeverity = (alert.severity || "info").toLowerCase() as AlertSeverity;
  const isAcked = isAlertAcknowledged(alert);
  const isSolved = isAlertSolved(alert);

  const getSeverityMeta = () => {
    switch (severity) {
      case "critical":
        return {
          color: "#DC2626",
          badgeBg: "#FEE2E2",
          badgeText: "#DC2626",
          icon: <AlertCircle size={13} color="#DC2626" />,
          label: "CRITICAL",
        };
      case "warning":
        return {
          color: "#D97706",
          badgeBg: "#FEF3C7",
          badgeText: "#D97706",
          icon: <AlertTriangle size={13} color="#D97706" />,
          label: "WARNING",
        };
      default:
        return {
          color: "#2563EB",
          badgeBg: "#DBEAFE",
          badgeText: "#2563EB",
          icon: <Info size={13} color="#2563EB" />,
          label: "INFO",
        };
    }
  };

  const meta = getSeverityMeta();

  // Parse timestamps safely
  const parseDate = (ts: string | Date | undefined) => {
    if (!ts) return new Date();
    try {
      const d = new Date(ts);
      return isNaN(d.getTime()) ? new Date() : d;
    } catch {
      return new Date();
    }
  };

  const alertDate = parseDate(alert.ts || alert.createdAt);

  const timeAgo = () => {
    try {
      return formatDistanceToNow(alertDate, { addSuffix: true });
    } catch {
      return "Recently";
    }
  };

  const formattedDateTime = () => {
    return formatAlertCardDate(alert.ts || alert.createdAt);
  };

  const performanceMetric = getPerformanceMetricName(alert.tag_name);
  const formattedVal = formatAlertValue(alert.alert_value);
  const formattedUnit = formatAlertUnit(alert.unit, alert.tag_name);

  return (
    <View style={[styles.card, isSolved && styles.cardSolved]}>
      {/* 1. Top Meta Row: Severity + Acknowledged/Solved + Timestamp + Dots */}
      <View style={styles.topRow}>
        <View style={styles.badgeGroup}>
          <View style={[styles.severityBadge, { backgroundColor: meta.badgeBg }]}>
            {meta.icon}
            <Text style={[styles.severityText, { color: meta.badgeText }]}>
              {meta.label}
            </Text>
          </View>

          {isAcked && !isSolved && (
            <View style={styles.ackBadge}>
              <CheckCircle2 size={12} color="#16A34A" />
              <Text style={styles.ackBadgeText}>ACKNOWLEDGED</Text>
            </View>
          )}

          {isSolved && (
            <View style={styles.solvedBadge}>
              <CheckCircle2 size={12} color="#059669" />
              <Text style={styles.solvedBadgeText}>SOLVED</Text>
            </View>
          )}
        </View>

        <View style={styles.topRight}>
          <View style={styles.timeWrapper}>
            <Clock size={13} color="#64748B" />
            <Text style={styles.timeAgoText}>{timeAgo()}</Text>
          </View>
          <TouchableOpacity
            style={styles.moreBtn}
            onPress={onPress}
            hitSlop={{ top: 8, bottom: 8, left: 8, right: 8 }}
          >
            <MoreVertical size={16} color="#64748B" />
          </TouchableOpacity>
        </View>
      </View>

      {/* 2. Panel Name */}
      <Text style={styles.panelName} numberOfLines={1}>
        {alert.panel_name || "Unknown Panel"}
      </Text>

      {/* 3. Performance Metric & Value Box */}
      <View style={styles.metricContainer}>
        <View style={styles.metricLeft}>
          <Text style={styles.metricLabel}>PERFORMANCE METRIC</Text>
          <Text style={styles.metricTag} numberOfLines={1}>
            {performanceMetric}
          </Text>
        </View>

        <View style={styles.metricRight}>
          <Text style={styles.metricLabel}>VALUE</Text>
          <Text style={styles.metricValue} numberOfLines={1}>
            {formattedVal}
            {formattedUnit ? (
              <Text style={styles.metricUnit}> {formattedUnit}</Text>
            ) : null}
          </Text>
        </View>
      </View>

      {/* 4. Alert Message */}
      <Text style={styles.messageText} numberOfLines={2}>
        {alert.message}
      </Text>

      {/* 5. Footer: Exact Timestamp & Action Buttons */}
      <View style={styles.footerRow}>
        <View style={styles.dateTimeWrapper}>
          <Calendar size={13} color="#64748B" />
          <Text style={styles.exactTimeText}>{formattedDateTime()}</Text>
        </View>

export function matchesActiveFilters(
  alert: AlertItem,
  filters: FilterState,
  searchQuery: string
): boolean {
  if (!alert) return false;

  // 1. Severity filter
  if (filters.severity && filters.severity !== "ALL") {
    const alertSev = (alert.severity || "").toLowerCase();
    if (alertSev !== filters.severity.toLowerCase()) {
      return false;
    }
  }

  // 2. Alert Message Key filter (Critical, Warning, Returned to Normal)
  if (filters.messageKey && filters.messageKey !== "ALL") {
    const msgKey = filters.messageKey.toUpperCase();
    const alertMsg = (alert.message || "").toLowerCase();
    const alertSev = (alert.severity || "").toLowerCase();
    if (msgKey === "CRITICAL") {
      if (!alertMsg.includes("crit") && alertSev !== "critical") return false;
    } else if (msgKey === "WARNING") {
      if (!alertMsg.includes("warn") && alertSev !== "warning") return false;
    } else if (msgKey === "NORMAL") {
      if (!alertMsg.includes("normal") && !alertMsg.includes("returned")) return false;
    }
  }

  // 3. Status filter (acknowledgement & lifecycle state)
  if (filters.status && filters.status !== "ALL") {
    const targetStatus = filters.status.toLowerCase();

    if (targetStatus === "open" || targetStatus === "unacknowledged") {
      if (isAlertAcknowledged(alert)) return false;
      if (isAlertSolved(alert)) return false;
    } else if (targetStatus === "acknowledged") {
      if (!isAlertAcknowledged(alert)) return false;
    } else if (targetStatus === "unsolved") {
      if (isAlertSolved(alert)) return false;
    } else if (targetStatus === "cleared" || targetStatus === "solved") {
      if (!isAlertSolved(alert)) return false;
    } else if (targetStatus === "solved_by_system") {
      if (!isAlertSolvedBySystem(alert)) return false;
    }
  }

  // 4. Resolution filter (Solved / Unsolved)
  if (filters.resolution && filters.resolution !== "ALL") {
    const targetRes = filters.resolution.toLowerCase();
    const isSolved = isAlertSolved(alert);
    if (targetRes === "solved" && !isSolved) return false;
    if (targetRes === "unsolved" && isSolved) return false;
  }

  // 5. Panel / Equipment filter (CRITICAL FIX: case-insensitive, name, ID, location, or device_id match)
  if (filters.panel && filters.panel !== "ALL" && filters.panel.trim() !== "") {
    const target = filters.panel.trim().toLowerCase();
    const pName = (alert.panel_name || "").trim().toLowerCase();
    const pId = (alert.panel_id || "").trim().toLowerCase();
    const dId = (alert.device_id || (alert as any).did || "").trim().toLowerCase();
    const location = (alert.panel_location || "").trim().toLowerCase();

    if (
      pName !== target &&
      pId !== target &&
      dId !== target &&
      location !== target &&
      !pName.includes(target)
    ) {
      return false;
    }
  }

  // 6. User filter (matches ack_user, solved_user, clear_user)
  if (filters.user && filters.user !== "ALL" && filters.user.trim() !== "") {
    const targetUser = filters.user.trim().toLowerCase();
    const alertUsers = [
      alert.ack_user,
      alert.ack_user_name,
      alert.ack_user_email,
      alert.clear_user,
      alert.solved_user,
      alert.solved_user_name,
      alert.solved_user_email,
      (alert as any).user,
    ]
      .map((x) => String(x || "").trim().toLowerCase())
      .filter(Boolean);

    if (!alertUsers.some((u) => u === targetUser || u.includes(targetUser))) {
      return false;
    }
  }

  // 7. Performance Metric / Tag filter
  if (filters.tag && filters.tag !== "ALL" && filters.tag.trim() !== "") {
    const targetTag = filters.tag.trim().toLowerCase();
    const tag = (alert.tag_name || "").trim().toLowerCase();
    const metricKey = (alert.metric_key || "").trim().toLowerCase();
    const humanName = getPerformanceMetricName(alert.tag_name).trim().toLowerCase();

    if (
      tag !== targetTag &&
      metricKey !== targetTag &&
      humanName !== targetTag &&
      !humanName.includes(targetTag)
    ) {
      return false;
    }
  }

  // 8. Occurrence Counter filter
  if (filters.counter && filters.counter !== "ALL" && filters.counter.trim() !== "") {
    const c = Number(alert.occurrence_counter ?? 0);
    if (filters.counter === "Under 5" && c >= 5) return false;
    if (filters.counter === "5+" && c < 5) return false;
    if (filters.counter === "10+" && c < 10) return false;
    if (filters.counter === "15+" && c < 15) return false;
  }

  // 9. Duration / Timer filter (active_accumulated_ms in milliseconds)
  if (filters.timer && filters.timer !== "ALL" && filters.timer.trim() !== "") {
    const ms = Number(alert.active_accumulated_ms ?? 0);
    const hours = ms / (1000 * 60 * 60);
    if (filters.timer === "Under 1 Hr" && hours >= 1) return false;
    if (filters.timer === "More than 1 Hr" && hours < 1) return false;
    if (filters.timer === "More than 5 Hr" && hours < 5) return false;
    if (filters.timer === "More than 10 Hr" && hours < 10) return false;
    if (filters.timer === "More than 24 Hr" && hours < 24) return false;
  }

  // 10. Search query filter
  if (searchQuery && searchQuery.trim()) {
    const q = searchQuery.trim().toLowerCase();
    const searchable = `${alert.panel_name || ""} ${alert.tag_name || ""} ${getPerformanceMetricName(alert.tag_name)} ${alert.device_id || ""} ${alert.message || ""}`.toLowerCase();
    if (!searchable.includes(q)) {
      return false;
    }
  }

  return true;
}

export const getTodayYmd = (): string => {
  const now = new Date();
  const IST_OFFSET_MS = 5.5 * 60 * 60 * 1000;
  const nowIst = new Date(now.getTime() + IST_OFFSET_MS);
  const y = nowIst.getUTCFullYear();
  const m = String(nowIst.getUTCMonth() + 1).padStart(2, "0");
  const d = String(nowIst.getUTCDate()).padStart(2, "0");
  return `${y}-${m}-${d}`;
};

  // Core REST API Fetch Function for LIVE ALERT STREAM (Lifecycle Driven)

  const fetchAlerts = useCallback(
    async (refresh: boolean = false) => {
      if (!isAuthenticated) return;

      // If another fetch is already running, queue refresh
      if (isFetchingRef.current) {
        if (refresh) {
          pendingRefreshRef.current = true;
          console.log(
            "[AlertContext] Refresh queued because another fetch is already running."
          );
        }
        return;
      }

      isFetchingRef.current = true;
      setFetchError(null);

      if (refresh) {
        setIsRefreshing(true);
      } else {
        setIsLoading(true);
      }

      const targetPage = refresh ? 1 : page;
      const requestId = ++listRequestIdRef.current;

      if (refresh || targetPage === 1) {
        backendCandidatePageRef.current = 1;
      }

      try {
    const { start, end } = getEffectiveAlertRange(filters);
    console.log("[LIST DATE RANGE]", {
  start,
  end,
  customStartDate: filters.customStartDate,
  customEndDate: filters.customEndDate,
});

        const queryParams: ListAlertsQuery = {
          severity:
            filters.severity !== "ALL"
              ? filters.severity.toLowerCase()
              : undefined,

          // Status filtering is lifecycle-driven and applied on frontend
          // after lifecycle decoration. DO NOT send legacy status to backend.
          status: undefined,

          search:
            searchQuery.trim()
              ? searchQuery.trim()
              : undefined,

          // CRITICAL: DO NOT send panel name (e.g. "Flat Drip Kabra Panel") to backend panel_id!
          // MongoDB alerts store panel_id as null, and panel_name as the display name.
          // Sending panel_id causes backend to return 0 results.
          // Frontend candidate gathering handles panel, user, metrics, timer, counter, and lifecycle.
          panel_id: undefined,

          tag_name: undefined,

          start,
          end,

          includeSummary: false,
        };

        const targetBatchSize = filters.pageSize || pageSize; // 20
        const hasClientFilter = Boolean(
          (filters.status && filters.status !== "ALL") ||
          (filters.resolution && filters.resolution !== "ALL") ||
          (filters.panel && filters.panel !== "ALL" && filters.panel.trim() !== "") ||
          (filters.tag && filters.tag !== "ALL" && filters.tag.trim() !== "") ||
          (filters.user && filters.user !== "ALL" && filters.user.trim() !== "") ||
          (filters.counter && filters.counter !== "ALL" && filters.counter.trim() !== "") ||
          (filters.timer && filters.timer !== "ALL" && filters.timer.trim() !== "") ||
          (filters.messageKey && filters.messageKey !== "ALL" && filters.messageKey.trim() !== "")
        );

        const collectedFiltered: AlertItem[] = [];
        let backendPage = backendCandidatePageRef.current;
        let backendHasMore = true;

        const existingIds = new Set<string>(
          (refresh || targetPage === 1)
            ? []
            : alertsRef.current.map((a) => a._id || a.alert_id || (a as any)?.id)
        );

        // Fetch candidate pages sequentially until enough matching lifecycle rows are gathered
        // OR backend hasMore becomes false
        while (backendHasMore) {
          const res = await alertApi.listAlerts({
            ...queryParams,
            page: backendPage,
            pageSize: hasClientFilter ? 50 : targetBatchSize,
          });

          const rawRows: AlertItem[] = res?.rows || [];
          backendHasMore = Boolean(res?.hasMore ?? (rawRows.length === (hasClientFilter ? 50 : targetBatchSize)));
          backendPage++;

          if (rawRows.length > 0) {
            const decoratedRows = await decorateAlertsWithLifecycle(rawRows);
            const matching = decoratedRows.filter((alert) =>
              matchesActiveFilters(alert, filters, searchQuery)
            );

            for (const item of matching) {
              const id = item._id || item.alert_id || (item as any)?.id;
              if (id && !existingIds.has(id)) {
                existingIds.add(id);
                collectedFiltered.push(item);
              } else if (!id) {
                collectedFiltered.push(item);
              }
            }
          }

          // If no client filter is active, 1 candidate page satisfies 1 frontend page
          if (!hasClientFilter) {
            break;
          }

          // If client filter is active, stop once we've collected enough matching rows
          if (collectedFiltered.length >= targetBatchSize || !backendHasMore || rawRows.length === 0) {
            break;
          }

          // Safety guard: max 20 candidate pages per load batch
          if (backendPage > backendCandidatePageRef.current + 20) {
            break;
          }
        }

        // Drop response if a newer filter/refresh request was launched
        if (requestId !== listRequestIdRef.current) {
          return;
        }

        backendCandidatePageRef.current = backendPage;

        if (refresh || targetPage === 1) {
          setAlerts(collectedFiltered);
          setPage(1);
        } else {
          setAlerts((prev) => [...prev, ...collectedFiltered]);
        }

        setHasMore(backendHasMore);
        setFetchError(null);
      } catch (err: any) {
        if (requestId !== listRequestIdRef.current) {
          return;
        }
        const status = err?.response?.status;
        let userMessage: string;

        if (!err?.response) {
          if (err?.code === "ECONNABORTED") {
            userMessage = "Alert service request timed out. Please retry.";
          } else {
            userMessage = "Unable to connect to alert service. Check your connection.";
          }
        } else if (status === 401) {
          userMessage = "Session expired. Please sign in again.";
        } else if (status === 403) {
          userMessage = "Access restricted. Alert service could not resolve your company.";
        } else if (status === 404) {
          userMessage = "Alert API route not found.";
        } else {
          userMessage = `Alert service error (${status}).`;
        }

        console.log("[AlertContext] fetchAlerts error:", { status, message: userMessage });
        setFetchError(userMessage);
      } finally {
        if (requestId === listRequestIdRef.current) {
          setIsLoading(false);
          setIsRefreshing(false);
        }
        isFetchingRef.current = false;

        // If a lifecycle mutation requested a refresh while another fetch was running, execute that refresh now
        if (pendingRefreshRef.current) {
          pendingRefreshRef.current = false;
          console.log("[AlertContext] Running queued lifecycle refresh.");
          setTimeout(() => {
            fetchAlerts(true);
          }, 0);
        }
      }
    },
    [isAuthenticated, page, filters, searchQuery]
  );

  const fetchAlertsRef = useRef(fetchAlerts);
  useEffect(() => {
    fetchAlertsRef.current = fetchAlerts;
  }, [fetchAlerts]);

  const fetchSummaryRef = useRef(fetchSummary);

  // Refetch when page or filters change
  useEffect(() => {

      fetchAlerts(page === 1);
    }
  }, [page, filters, searchQuery, isAuthenticated]);

  // Load next page
  const loadMore = useCallback(async () => {
    if (!hasMore || isLoading || isRefreshing || isFetchingRef.current) return;
    setPage((prev) => prev + 1);
  }, [hasMore, isLoading, isRefreshing]);

const acknowledgeAlert = async (id: string, note?: string) => {
  if (!id) return;

  try {
    const payload: {
      action: "ack" | "unack";
      note?: string;
    } = {
      action: "ack",
    };

    if (note && note.trim()) {
      payload.note = note.trim();
    }

    console.log("[MOBILE ACK REQUEST]", {
      alertId: id,
      endpoint: `/api/v1/alerts/${id}/ack`,
      method: "PATCH",
      noteProvided: Boolean(note?.trim()),
      hasToken: Boolean(accessToken),
    });

    const res = await alertApi.acknowledgeAlert(
      id,
      payload
    );

    console.log("[MOBILE ACK RESPONSE]", {
      status: 200,
      keys:
        res && typeof res === "object"
          ? Object.keys(res)
          : [],
    });

    if (res?.data) {
      const updated = res.data;

      setAlerts((prev) =>
        prev
          .map((item) => {
            const matchesId =
              String(item.alert_id || "") === String(id) ||
              String(item._id || "") === String(id);

            return matchesId
              ? ({
                  ...item,
                  ...updated,

                  status: "acknowledged" as AlertStatus,
                  lifecycle_acknowledged: true,
                  acknowledged: true,
                  bulk_ack_status: null,
                } as AlertItem)
              : item;
          })
          .filter((item): item is AlertItem =>
            matchesActiveFilters(item, activeFiltersRef.current, searchQueryRef.current)
          )
      );
    } else {
      await fetchAlerts(false);
    }

    // Preserve the real _id and alert_id separately.
    const currentObj = alerts.find(
      (a) =>
        String(a.alert_id || "") === String(id) ||
        String(a._id || "") === String(id)
    );

    const ackedItem = {
      ...(currentObj || {}),
      ...(res?.data || {}),

      _id:
        res?.data?._id ??
        currentObj?._id,

      alert_id:
        res?.data?.alert_id ??
        currentObj?.alert_id,

      lifecycle_acknowledged: true,

        res?.data?.ack_note ??

      ackedItem,

      status: err?.response?.status,
      data: err?.response?.data,
      message: err?.message,
      url: err?.config?.url,
      method: err?.config?.method,
    });

    throw err;
  }

      (a) =>
        String(a.alert_id || "") === String(id) ||
        String(a._id || "") === String(id)
    );

  if (!targetAlert) {

    );
  }

  // Always use the lifecycle API for:
  // This Alert / Today / Last 1 Week / All Previous.
  // Do NOT fall back to the legacy clearAlert API.
  const result = await applyLifecycleAction(
    targetAlert,
    "solve",
    scope,
    resolutionNote
  );

  // Restart list from page 1 after lifecycle mutation.
  setPage(1);

  // fetchAlerts() already decorates the returned rows
  // using lifecycle state.
  await fetchAlerts(true);

  await fetchSummary();

  useEffect(() => {
    if (!isAuthenticated || !accessToken) {
      // Clear per-device deduplication cache on logout so a newly logged-in user on this phone receives alerts
      processedAlertIdsRef.current.clear();
      return;
    }

    // Ensure socket is actively connected with CURRENT user's accessToken
    socketService.connect(accessToken).catch((e) =>
      console.warn("[AlertContext] Socket connect:", e?.message)
    );

    const unsubStatus = socketService.onStatusChange((status, detail) => {
      setConnectionStatus(status);
      setConnectionDetail(detail);
      if (status === "unauthorized") {
        console.warn("[AlertContext] Socket unauthorized (expired token) â†’ auto logout");
        logout();
      }
    });

    // Handle genuinely NEW realtime alert from live stream
    const unsubAlertNew = socketService.onAlertNew(async (newAlert: AlertItem) => {
      // Preferred unique alert ID order: 1. alert._id -> 2. alert.alert_id -> 3. alert.id
      const primaryAlertId = newAlert._id || newAlert.alert_id || (newAlert as any)?.id;
      const stableAlertId = primaryAlertId ? String(primaryAlertId).trim() : "";
      // Dedup key: uses unique alert ID, or composite key (DID + timestamp + metric + severity). NEVER DID alone.
      const dedupKey = getAlertDeduplicationKey(newAlert);
      const severity = normalizeSeverity(newAlert.severity as string);

      // Stage 1: Log Socket event arrival BEFORE any notification filtering or deduplication
      console.log("[PowerIQ Socket Alert Received]", {
        alertId: stableAlertId || dedupKey || "unknown",
        dedupKey: dedupKey || "unkeyed",
        severity,
        panel: newAlert.panel_name || "unknown",
        metric: newAlert.tag_name || (newAlert as any)?.tag || (newAlert as any)?.metric || (newAlert as any)?.parameter || "unknown",
      });

      // Deduplicate strictly using unique alert ID or composite key (NEVER DID alone)
      // If dedupKey is empty (no unique ID and insufficient metadata), do not suppress
      const isAlreadyHandled =
        (dedupKey && (processedAlertIdsRef.current.has(dedupKey) || isAlertProcessed(dedupKey))) ||
        (stableAlertId && isAlertProcessed(stableAlertId));

      if (isAlreadyHandled) {
        console.log("[AlertContext] Duplicate socket alert ignored on this device:", dedupKey || stableAlertId);
        return;
      }

      if (dedupKey) {
        processedAlertIdsRef.current.add(dedupKey);
        markAlertProcessed(dedupKey);
        if (stableAlertId) markAlertProcessed(stableAlertId);
        if (processedAlertIdsRef.current.size > 200) {
          const first = processedAlertIdsRef.current.values().next().value;
          if (first) processedAlertIdsRef.current.delete(first);
        }
      } else if (stableAlertId) {
        markAlertProcessed(stableAlertId);
      }

      // Canonical role filtering: admin, super admin, production, maintenance, engineer, and owner are eligible
      const currentRole = userRoleRef.current;
      const isEligible = isNotificationEligibleRole(currentRole);
      const isOwner = isOwnerRole(currentRole);

      logPowerIQRoleDebug(user?.name, currentRole);

      if (isEligible) {
        // If owner, check if owner notifications are enabled in settings
        if (isOwner) {
          const ownerEnabled = await getOwnerNotificationsEnabled();
          if (!ownerEnabled) {
            console.log("[AlertContext] Realtime alert received for Owner â€” notifications disabled in settings.");
            setLatestIncomingAlert(newAlert);
            return;
          }
        }

        // Stage 2: Role Eligible
        console.log("[PowerIQ Role Eligible]", {
          userRole: currentRole,
          canonicalRole: normalizeRole(currentRole),
          eligible: true,
          isOwner,
        });

        // Stage 3: System Notification Scheduling
        try {
          await notifyForRealtimeAlert(newAlert, currentRole);
        } catch (error: any) {
          console.warn("[AlertContext] Local notification failed:", error?.message);
        }

        // Stage 4 & 5: Sound Playback (For Owner: always plays poweriq_alert.wav)
        const severitySoundEnabled = await getSeveritySoundPreference(severity);

        if (severity === "critical" && !isOwner) {
          console.log("[PowerIQ Critical Sound Starting]", { alertId: stableAlertId || dedupKey });
        }

        if (severitySoundEnabled || isOwner) {
          await soundService.playSoundForNotification(severity, "socket", isOwner);
          if (severity === "critical" && !isOwner) {
            console.log("[PowerIQ Critical Sound Started]", { alertId: stableAlertId || dedupKey });
          }
        } else {
          console.log("[PowerIQ Sound Skipped by User Preference]", {
            severity,
            alertId: stableAlertId || dedupKey,
          });
        }

        // Show in-app banner toast
        setLatestIncomingAlert(newAlert);
      } else {
        console.log(
          `[AlertContext] Realtime alert ${stableAlertId || dedupKey} received for role "${currentRole}" â€” notification skipped.`
        );
      }

      // Single source of truth: refresh feed and summary via lifecycle decoration pipeline
      // without manually mutating counters or inserting undecorated items
      await fetchAlertsRef.current(true);
      await fetchSummaryRef.current();
    });

    // Handle alert update / ack broadcast as a refresh signal
    const unsubAlertUpdated = socketService.onAlertUpdated(async () => {
      try {
        await fetchAlertsRef.current(true);
        await fetchSummaryRef.current();
      } catch (err) {
        console.warn("[AlertContext] socket onAlertUpdated refresh error:", err);
      }
    });

    // Handle native push notifications (when app is in foreground/background)
    const unsubPush = setupPushNotificationListeners(
      async (pushAlert) => {
        setLatestIncomingAlert(pushAlert);
        await fetchAlertsRef.current(true);
        await fetchSummaryRef.current();
      }
    );

    return () => {
      unsubStatus();
      unsubAlertNew();
      unsubAlertUpdated();
      unsubPush();
    };
  }, [isAuthenticated, accessToken]);

  const dismissToast = () => setLatestIncomingAlert(null);

  const setFilters = (
    updater: Partial<FilterState> | ((prev: FilterState) => FilterState)
  ) => {
    setFiltersState((prev) => {
      const next = typeof updater === "function" ? updater(prev) : { ...prev, ...updater };
      return next;
    });
    setPage(1);
  };

  const resetFilters = () => {
    setFiltersState(DEFAULT_FILTERS);
    setSearchQuery("");
    setPage(1);
  };

  return (
    <AlertContext.Provider
      value={{
        alerts,
        counters,
        totalAlertsCount,
        isLoading,
        isRefreshing,
        hasMore,
        connectionStatus,
        connectionDetail,
        searchQuery,
        filters,
        activeFilterCount,
        latestIncomingAlert,
        fetchError,
        fetchAlerts: async () => {
          await Promise.all([fetchAlerts(true), fetchSummary()]);
        },
        loadMore,
        acknowledgeAlert,
        unacknowledgeAlert,
        acknowledgeAlertWithScope,
        unacknowledgeAlertWithScope,
        resolveAlertWithScope,
        unsolveAlertWithScope,
        setSearchQuery: (q) => {
          setSearchQuery(q);
          setPage(1);
        },
        setFilters,
        resetFilters,
        dismissToast,
      }}
    >
      {children}
    </AlertContext.Provider>

/**
 * Custom hook providing direct access to the live alert stream,

 */
export const useLiveAlerts = (options?: {
  onNewAlert?: (alert: AlertItem) => void;
  onAlertUpdated?: (alert: AlertItem) => void;
}) => {
  const { accessToken } = useAuth();
  const alertContext = useAlerts();

  useEffect(() => {
    if (!options?.onNewAlert && !options?.onAlertUpdated) return;

    let unsubNew: (() => void) | undefined;
    let unsubUp: (() => void) | undefined;

    if (options.onNewAlert) {
      unsubNew = socketService.onAlertNew(options.onNewAlert);
    }
    if (options.onAlertUpdated) {
      unsubUp = socketService.onAlertUpdated(options.onAlertUpdated);
    }

    return () => {
      unsubNew?.();
      unsubUp?.();
    };
  }, [options?.onNewAlert, options?.onAlertUpdated]);

  const reconnect = async () => {
    if (accessToken) {
      await socketService.connect(accessToken);
    }
  };

  return {
    ...alertContext,
    reconnect,
    isConnected: alertContext.connectionStatus === "connected",
    isReconnecting: alertContext.connectionStatus === "reconnecting",
  };

const mergeAlertDetails = (

    _id: incoming._id ?? previous._id,

    // Keep canonical alert/device information â€” NEVER overwrite valid previous data with empty or "-"
    panel_name: isMeaningful(incoming.panel_name)

      : isMeaningful(previous.device_id)
      ? previous.device_id
      : incoming.device_id,

    // Keep metric information

    // Keep recorded value

const getResolvedByName = (alert?: AlertItem | null): string => {
  if (!alert) return "Not specified";

  const candidates = [
    alert.solved_user_name,
    alert.solved_user,
    alert.solved_by && alert.solved_by.toLowerCase() === "system"
      ? "System"
      : alert.solved_by,
    alert.clear_user,
  ];

  for (const candidate of candidates) {
    if (typeof candidate === "string") {
      const trimmed = candidate.trim();
      if (trimmed && trimmed.toLowerCase() !== "operator") {
        if (trimmed.toLowerCase() === "system") {
          return "System";
        }
        return trimmed;
      }
    }
  }

  return "Not specified";
};

            style={styles.heroCardPlantBackdrop}
            resizeMode="cover"
          />
          <View style={styles.heroContent}>
            <Text style={styles.heroTitle}>Reliable Operations</Text>
            <Text style={styles.heroTitle}>Brighter Tomorrow</Text>
            <Text style={styles.heroSubtitle}>Stay informed. Stay in control.</Text>
          </View>
          
        </LinearGradient>
      </View>

      {/* â”€â”€ 3. Alert Summary Section â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€ */}
      <View style={styles.sectionHeaderRow}>
        <Text style={styles.sectionHeading}>Alert Summary</Text>
        <TouchableOpacity
          style={styles.dateSelectorBtn}
          onPress={() => setIsFilterModalOpen(true)}
          activeOpacity={0.7}
        >
          <Text style={styles.dateSelectorText}>
            {formatDisplayDateRange(filters.customStartDate, filters.customEndDate)}
          </Text>
          <ChevronDown size={14} color="#64748B" />
        </TouchableOpacity>
      </View>

      {/* Featured Total Alerts Card */}
      <TouchableOpacity
        style={[
          styles.totalAlertsCard,
          filters.severity === "ALL" && filters.status === "ALL" && styles.cardActiveHighlight,
        ]}
        onPress={() => setFilters((prev) => ({ ...prev, severity: "ALL", status: "ALL" }))}
        activeOpacity={0.8}
      >
        <View style={styles.totalAlertsLeft}>
          <View style={styles.totalAlertsIconWrap}>
            <Activity size={22} color="#0284C7" />
          </View>
          <View style={styles.totalAlertsData}>
            <Text style={styles.totalAlertsLabel}>Total Alerts</Text>
            <View style={styles.totalAlertsValueRow}>
              <Text style={styles.totalAlertsCount}>{totalAlertsCount}</Text>
              <BarChart2 size={24} color="#94A3B8" style={{ marginLeft: 16 }} />
            </View>
          </View>
        </View>

        <View style={styles.totalAlertsDivider} />

        <View style={styles.totalAlertsRight}>
          <Text style={styles.totalAlertsTrend}>
            {counters.pending_critical + counters.pending_warning}
          </Text>
          <Text style={styles.totalAlertsSub}>pending alerts</Text>
        </View>
      </TouchableOpacity>

      {/* 2x2 Quick Action Cards Grid */}
      <View style={styles.metricsGrid}>
        {/* Pending Critical */}
        <TouchableOpacity
          style={[
            styles.gridCard,
            filters.severity === "critical" && filters.status === "open" && styles.gridCardActive,
          ]}
          onPress={() =>
            setFilters((prev) => {
              const isActive = prev.severity === "critical" && prev.status === "open";
              return {
                ...prev,
                severity: isActive ? "ALL" : "critical",
                status: isActive ? "ALL" : "open",
              };
            })
          }
          activeOpacity={0.8}
        >
          <View style={styles.gridCardLeft}>
            <View style={[styles.gridCardIconCircle, { backgroundColor: "#FEE2E2" }]}>
              <AlertCircle size={18} color="#DC2626" />
            </View>
            <View style={styles.gridCardInfo}>
              <Text style={styles.gridCardTitle}>Pending Critical</Text>
              <Text style={styles.gridCardCount}>{counters.pending_critical}</Text>
            </View>
          </View>
          <ChevronRight size={16} color="#94A3B8" />
        </TouchableOpacity>

        {/* Pending Warning */}
        <TouchableOpacity
          style={[
            styles.gridCard,
            filters.severity === "warning" && filters.status === "open" && styles.gridCardActive,
          ]}
          onPress={() =>
            setFilters((prev) => {
              const isActive = prev.severity === "warning" && prev.status === "open";
              return {
                ...prev,
                severity: isActive ? "ALL" : "warning",
                status: isActive ? "ALL" : "open",
              };
            })
          }
          activeOpacity={0.8}
        >
          <View style={styles.gridCardLeft}>
            <View style={[styles.gridCardIconCircle, { backgroundColor: "#FEF3C7" }]}>
              <AlertTriangle size={18} color="#D97706" />
            </View>
            <View style={styles.gridCardInfo}>
              <Text style={styles.gridCardTitle}>Pending Warning</Text>
              <Text style={styles.gridCardCount}>{counters.pending_warning}</Text>
            </View>
          </View>
          <ChevronRight size={16} color="#94A3B8" />
        </TouchableOpacity>

        {/* Solved */}
        <TouchableOpacity
          style={[
            styles.gridCard,
            filters.status === "solved" && filters.severity === "ALL" && styles.gridCardActive,
          ]}
          onPress={() =>
            setFilters((prev) => {
              const isActive = prev.status === "solved" && prev.severity === "ALL";
              return {
                ...prev,
                severity: "ALL", // Always clear severity to ALL so ALL solved alerts (both critical & warning) show
                status: isActive ? "ALL" : "solved",
              };
            })
          }
          activeOpacity={0.8}
        >
          <View style={styles.gridCardLeft}>
            <View style={[styles.gridCardIconCircle, { backgroundColor: "#DCFCE7" }]}>
              <CheckCircle2 size={18} color="#16A34A" />
            </View>
            <View style={styles.gridCardInfo}>
              <Text style={styles.gridCardTitle}>Total Solved</Text>
              <Text style={styles.gridCardCount}>
                {counters.all_solved ?? 0}
              </Text>
            </View>
          </View>
          <ChevronRight size={16} color="#94A3B8" />
        </TouchableOpacity>

        {/* Solved by System */}
        <TouchableOpacity
          style={[
            styles.gridCard,
            filters.status === "solved_by_system" && filters.severity === "ALL" && styles.gridCardActive,
          ]}
          onPress={() =>
            setFilters((prev) => {
              const isActive = prev.status === "solved_by_system" && prev.severity === "ALL";
              return {
                ...prev,
                severity: "ALL", // Always clear severity to ALL so ALL system-solved alerts show
                status: isActive ? "ALL" : "solved_by_system",
              };
            })
          }
          activeOpacity={0.8}
        >
          <View style={styles.gridCardLeft}>
            <View style={[styles.gridCardIconCircle, { backgroundColor: "#E0F2FE" }]}>
              <Settings size={18} color="#0284C7" />
            </View>
            <View style={styles.gridCardInfo}>
              <Text style={styles.gridCardTitle}>Solved by System</Text>
              <Text style={styles.gridCardCount}>
                {counters.solved_by_system ?? 0}
              </Text>
            </View>
          </View>
          <ChevronRight size={16} color="#94A3B8" />
        </TouchableOpacity>
      </View>

      {/* â”€â”€ 4. Search & Filter Bar (Shown in Alerts mode or when search active) â”€â”€ */}
      {(activeTab === "alerts" || searchQuery.trim() !== "" || activeFilterCount > 0) && (
        <View style={styles.searchSection}>
          <View style={styles.searchBar}>
            <Search size={16} color="#64748B" style={{ marginRight: 8 }} />
            <TextInput
              style={styles.searchInput}
              placeholder="Search panel, tag, device or message..."
              placeholderTextColor="#94A3B8"
              value={searchInput}
              onChangeText={setSearchInput}
              onSubmitEditing={handleSearchSubmit}
              returnKeyType="search"
            />
            {searchInput.length > 0 ? (
              <TouchableOpacity onPress={handleClearSearch} style={styles.clearSearchBtn}>
                <X size={15} color="#64748B" />
              </TouchableOpacity>
            ) : (
              <TouchableOpacity
                onPress={() => setIsFilterModalOpen(true)}
                style={styles.clearSearchBtn}
              >
                <SlidersHorizontal size={15} color="#64748B" />
              </TouchableOpacity>
            )}
          </View>

          {/* Active Filter Chips */}
          {activeFilterCount > 0 && (
            <ScrollView
              horizontal
              showsHorizontalScrollIndicator={false}
              contentContainerStyle={styles.activeChipsContent}
            >
              {Boolean((filters.customStartDate && filters.customStartDate !== getTodayYmd()) || (filters.customEndDate && filters.customEndDate !== getTodayYmd())) && (
                <TouchableOpacity
                  style={styles.chip}
                  onPress={() =>
                    setFilters({
                      customStartDate: getTodayYmd(),
                      customEndDate: getTodayYmd(),
                    })
                  }
                >
                  <Text style={styles.chipText}>
                    {formatDisplayDateRange(filters.customStartDate, filters.customEndDate)}
                  </Text>
                  <X size={12} color="#475569" />
                </TouchableOpacity>
              )}

              {filters.severity !== "ALL" && (
                <TouchableOpacity
                  style={styles.chip}
                  onPress={() => setFilters({ severity: "ALL" })}
                >
                  <Text style={styles.chipText}>Severity: {filters.severity}</Text>
                  <X size={12} color="#475569" />
                </TouchableOpacity>
              )}

              {filters.status !== "ALL" && (
                <TouchableOpacity
                  style={styles.chip}
                  onPress={() => setFilters({ status: "ALL" })}
                >
                  <Text style={styles.chipText}>Status: {filters.status}</Text>
                  <X size={12} color="#475569" />
                </TouchableOpacity>
              )}

              {filters.panel !== "" && filters.panel !== "ALL" && (
                <TouchableOpacity
                  style={styles.chip}
                  onPress={() => setFilters({ panel: "" })}
                >
                  <Text style={styles.chipText}>Panel: {filters.panel}</Text>
                  <X size={12} color="#475569" />
                </TouchableOpacity>
              )}

              <TouchableOpacity style={styles.clearAllBtn} onPress={resetFilters}>
                <RotateCcw size={11} color="#DC2626" />
                <Text style={styles.clearAllBtnText}>Clear All</Text>
              </TouchableOpacity>
            </ScrollView>
          )}
        </View>
      )}

      {/* â”€â”€ 5. Recent Alerts Header â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€ */}
      <View style={[styles.sectionHeaderRow, { marginTop: 26, marginBottom: 12 }]}>
        <Text style={styles.sectionHeading}>Recent Alerts</Text>
        <TouchableOpacity
          style={styles.viewAllBtn}
          onPress={() => {
            if (activeTab === "alerts") {
              setActiveTab("home");
            } else {
              setActiveTab("alerts");
            }
          }}
          activeOpacity={0.7}
        >
          <Text style={styles.viewAllText}>
            {activeTab === "alerts" ? "Show Summary" : "View All"}
          </Text>
          <ChevronRight size={14} color="#64748B" />
        </TouchableOpacity>
      </View>
    </View>
  );

    if (isLoading && !isRefreshing) {
      return (
        <View style={styles.emptyContainer}>
          <ActivityIndicator size="large" color="#2563EB" />
          <Text style={styles.emptySubtitle}>Loading live alerts...</Text>
        </View>
      );
    }

    if (fetchError) {
      return (
        <View style={styles.emptyContainer}>
          <View style={[styles.emptyIconCircle, { backgroundColor: "#FEE2E2" }]}>
            <BellOff size={28} color="#DC2626" />
          </View>
          <Text style={[styles.emptyTitle, { color: "#DC2626" }]}>Unable to Load Alerts</Text>
          <Text style={styles.emptySubtitle}>{fetchError}</Text>
          <TouchableOpacity style={styles.retryBtn} onPress={() => fetchAlerts()}>
            <Text style={styles.retryBtnText}>Retry Connection</Text>
          </TouchableOpacity>
        </View>
      );
    }

    return (
      <View style={styles.emptyContainer}>
        <View style={styles.emptyIconCircle}>
          <CheckCircle2 size={28} color="#16A34A" />
        </View>
        <Text style={styles.emptyTitle}>All Systems Normal</Text>
        <Text style={styles.emptySubtitle}>
          No active alarms match your current filters or selected timeframe.
        </Text>
      </View>
    );
  };

  const renderFooter = () => {
    if (!hasMore || alerts.length === 0) return <View style={{ height: 80 }} />;
    return (
      <View style={styles.footerLoader}>
        <ActivityIndicator size="small" color="#2563EB" />
      </View>
    );
  };

  return (
    <View style={[styles.container, { paddingTop: insets.top }]}>
      <StatusBar style="dark" />

      {/* Real-time In-App Notification Toast Banner */}
      <AlertToast
        alert={latestIncomingAlert}
        onPress={(alert) =>
          navigation.navigate("AlertDetail", {
            alert,
            alertId: alert._id || alert.alert_id,
          })
        }
        onDismiss={dismissToast}
      />

      {/* Main Alert Feed */}
      <FlatList
        data={alerts}
        keyExtractor={(item) => item._id || item.alert_id || String(Math.random())}
        renderItem={renderItem}
        ListHeaderComponent={renderHeader}
        ListEmptyComponent={renderEmpty}
        ListFooterComponent={renderFooter}
        onEndReached={loadMore}
        onEndReachedThreshold={0.3}
        refreshControl={
          <RefreshControl
            refreshing={isRefreshing}
            onRefresh={() => fetchAlerts(true)}
            colors={["#2563EB"]}
            tintColor="#2563EB"
          />
        }
        contentContainerStyle={styles.listContent}
        showsVerticalScrollIndicator={false}
      />

      {/* â”€â”€ Fixed Bottom Navigation Bar â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€ */}
      <View style={[styles.bottomNav, { paddingBottom: Math.max(insets.bottom, 12) }]}>
        <TouchableOpacity
          style={styles.bottomNavItem}
          onPress={() => setActiveTab("home")}
          activeOpacity={0.7}
        >
          <Home size={22} color={activeTab === "home" ? "#2563EB" : "#64748B"} />
          <Text
            style={[
              styles.bottomNavLabel,
              activeTab === "home" && styles.bottomNavLabelActive,
            ]}
          >
            Home
          </Text>
        </TouchableOpacity>

        <TouchableOpacity
          style={styles.bottomNavItem}
          onPress={() => setActiveTab("alerts")}
          activeOpacity={0.7}
        >
          <View>
            <Bell size={22} color={activeTab === "alerts" ? "#2563EB" : "#64748B"} />
            {totalAlertsCount > 0 && <View style={styles.bottomNavBadgeDot} />}
          </View>
          <Text
            style={[
              styles.bottomNavLabel,
              activeTab === "alerts" && styles.bottomNavLabelActive,
            ]}
          >
            Alerts
          </Text>
        </TouchableOpacity>

        <TouchableOpacity
          style={styles.bottomNavItem}
          onPress={() => {
            setIsFilterModalOpen(true);
          }}
          activeOpacity={0.7}
        >
          <Factory size={22} color={activeTab === "plants" ? "#2563EB" : "#64748B"} />
          <Text
            style={[
              styles.bottomNavLabel,
              activeTab === "plants" && styles.bottomNavLabelActive,
            ]}
          >
            Plants
          </Text>
        </TouchableOpacity>

        <TouchableOpacity
          style={styles.bottomNavItem}
          onPress={() => {
            setIsFilterModalOpen(true);
          }}
          activeOpacity={0.7}
        >
          <BarChart3 size={22} color={activeTab === "reports" ? "#2563EB" : "#64748B"} />
          <Text
            style={[
              styles.bottomNavLabel,
              activeTab === "reports" && styles.bottomNavLabelActive,
            ]}
          >
            Reports
          </Text>
        </TouchableOpacity>

        <TouchableOpacity
          style={styles.bottomNavItem}
          onPress={() => {
            if (isOwner) {
              navigation.navigate("Settings");
            } else {
              setIsMenuModalOpen(true);
            }
          }}
          activeOpacity={0.7}
        >
          <User size={22} color={activeTab === "profile" ? "#2563EB" : "#64748B"} />
          <Text
            style={[
              styles.bottomNavLabel,
              activeTab === "profile" && styles.bottomNavLabelActive,
            ]}
          >
            Profile
          </Text>
        </TouchableOpacity>
      </View>

      {/* â”€â”€ Hamburger Menu Modal / Drawer Sheet â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€ */}
      <Modal
        visible={isMenuModalOpen}
        transparent
        animationType="fade"
        onRequestClose={() => setIsMenuModalOpen(false)}
      >
        <TouchableOpacity
          style={styles.menuModalOverlay}
          activeOpacity={1}
          onPress={() => setIsMenuModalOpen(false)}
        >
          <View style={[styles.menuSheet, { paddingBottom: Math.max(insets.bottom, 24) }]}>
            <View style={styles.menuSheetHandle} />

            <View style={styles.menuUserProfile}>
              <View style={styles.menuAvatar}>
                <Text style={styles.menuAvatarText}>{userInitials}</Text>
              </View>
              <View style={styles.menuUserInfo}>
                <Text style={styles.menuUserName} numberOfLines={1}>
                  {userDisplayName}
                </Text>
                <View style={styles.menuRoleBadge}>
                  <Text style={styles.menuRoleText}>{displayRole}</Text>
                </View>
              </View>
            </View>

            <View style={styles.menuDivider} />

            <ConnectionStatus status={connectionStatus} detail={connectionDetail} />

            <View style={styles.menuDivider} />

            {/* Owner Settings button */}
            {isOwner && (
              <TouchableOpacity
                style={styles.menuActionItem}
                onPress={() => {
                  setIsMenuModalOpen(false);
                  navigation.navigate("Settings");
                }}
              >
                <Settings size={18} color="#0F172A" />
                <Text style={styles.menuActionText}>Notification & Alert Settings</Text>
                <ChevronRight size={16} color="#94A3B8" style={{ marginLeft: "auto" }} />
              </TouchableOpacity>
            )}

            <TouchableOpacity
              style={styles.menuActionItem}
              onPress={() => {
                setIsMenuModalOpen(false);
                setIsFilterModalOpen(true);
              }}
            >
              <SlidersHorizontal size={18} color="#0F172A" />
              <Text style={styles.menuActionText}>Filter & Scope Settings</Text>
              <ChevronRight size={16} color="#94A3B8" style={{ marginLeft: "auto" }} />
            </TouchableOpacity>

            <TouchableOpacity
              style={[styles.menuActionItem, { marginTop: 12 }]}
              onPress={handleLogout}
            >
              <LogOut size={18} color="#EF4444" />
              <Text style={[styles.menuActionText, { color: "#EF4444", fontWeight: "700" }]}>
                Sign Out
              </Text>
            </TouchableOpacity>
          </View>
        </TouchableOpacity>
      </Modal>

      {/* Filter Bottom Sheet Modal */}
      <FilterModal
        visible={isFilterModalOpen}
        onClose={() => setIsFilterModalOpen(false)}
        availablePanels={availablePanels}
        availableUsers={availableUsers}
        availableMetrics={availableMetrics}
        currentFilters={filters}
        totalCount={totalAlertsCount}
        onApply={(newFilters) => setFilters(newFilters)}
        onReset={resetFilters}
      />

      {/* Lifecycle Acknowledge Modal */}
      <AcknowledgeModal
        visible={!!selectedAlertForAck}
        onClose={() => setSelectedAlertForAck(null)}
        alert={selectedAlertForAck}
        onConfirm={handleConfirmAck}
      />

      {/* Lifecycle Resolve Modal */}
      <ResolveModal
        visible={!!selectedAlertForResolve}
        onClose={() => setSelectedAlertForResolve(null)}
        alert={selectedAlertForResolve}
        onConfirm={handleConfirmResolve}
      />
    </View>
  );
};

export function isAlertSolved(alert: AlertItem | null | undefined): boolean {
  if (!alert) return false;

  // 1. Explicit lifecycle solved state is PRIMARY
  if ((alert as any)?.solved !== undefined && (alert as any)?.solved !== null) {
    return (
      (alert as any).solved === true ||
      String((alert as any).solved).toLowerCase() === "true"
    );
  }

  // 2. Fallback to legacy Alerts service fields
  return (
    Boolean(alert.is_cleared) ||
    (alert.status || "").toLowerCase() === "cleared"
  );
}

export function isAlertSolvedBySystem(alert: AlertItem | null | undefined): boolean {
  if (!alert) return false;
  if (!isAlertSolved(alert)) return false;

  const solvedBy = String((alert as any)?.solved_by || (alert as any)?.solvedBy || "").toLowerCase();
  if (solvedBy === "system") return true;

  const resMethod = String(alert.resolution_method || (alert as any)?.solve_reason || "").toLowerCase();
  if (resMethod.includes("system") || resMethod.includes("auto")) return true;

  const clearUser = String(alert.clear_user || "").toLowerCase();
  const solvedUser = String(
    alert.solved_user ||
    (alert as any)?.solved_user_name ||
    (alert as any)?.solvedUser ||
    ""
  ).toLowerCase();

  if (clearUser.includes("system") || solvedUser.includes("system")) return true;

  // If solved without a human user assigned, it was auto-resolved by system
  const hasHumanUser = Boolean(
    (alert as any)?.solved_user_id ||
    (alert as any)?.solvedUserId ||
    (alert as any)?.clear_user_id ||
    (solvedUser && solvedUser !== "system" && solvedUser !== "null" && solvedUser !== "undefined")
  );

  return !hasHumanUser;
}

export function isAlertAcknowledged(alert: AlertItem | null | undefined): boolean {
  if (!alert) return false;

  // 1. Explicit lifecycle acknowledgement state is PRIMARY
  if (alert.lifecycle_acknowledged !== undefined && alert.lifecycle_acknowledged !== null) {
    return (
      alert.lifecycle_acknowledged === true ||
      String(alert.lifecycle_acknowledged).toLowerCase() === "true"
    );
  }
  if ((alert as any)?.acknowledged !== undefined && (alert as any)?.acknowledged !== null) {
    return (
      (alert as any).acknowledged === true ||
      String((alert as any).acknowledged).toLowerCase() === "true"
    );
  }

  // 2. Fallback to legacy Alerts service fields for alerts without lifecycle records
  const status = (alert.status || "open").toLowerCase();
  const bulkStatus = (alert.bulk_ack_status || "").toLowerCase();
  const effectiveStatus = bulkStatus || status;
  return effectiveStatus === "acknowledged";
}

export function getLifecycleDisplayState(alert: AlertItem | null | undefined): "solved" | "acknowledged" | "open" {
  if (!alert) return "open";
  const solved = isAlertSolved(alert);
  const acknowledged = isAlertAcknowledged(alert);

  if (solved) {
    return "solved";
  }

  if (acknowledged) {
    return "acknowledged";
  }

  return "open";
}

/**
 * Resolves the unique deduplication key for an alert.
 *
 * Strict priority order:
 *  1. alert._id
 *  2. alert.alert_id
 *  3. alert.id
 *
 * DID is NEVER used alone as a deduplication key.
 * Multiple distinct alerts can originate from the same device (DID), such as:
 *  - Power Factor Critical
 *  - Current Imbalance Critical
 *  - THD Critical
 * All of these must independently produce notifications/sirens.
 *
 * Fallback Composite Key (ONLY when primary IDs are absent):
 *  composite_${did}_${timestamp}_${metric/tag}_${severity}
 */
export function getAlertDeduplicationKey(alert: AlertItem | null | undefined): string {
  if (!alert) return "";

  // 1. Primary Unique Alert IDs
  const primaryId = alert._id || alert.alert_id || (alert as any)?.id;
  if (primaryId) {
    return String(primaryId).trim();
  }

  // 2. Fallback Composite Key (ONLY when primary IDs are absent; NEVER DID alone)
  const did = alert.device_id || (alert as any)?.did || "";
  const timestamp =
    alert.ts ||
    alert.createdAt ||
    (alert as any)?.timestamp ||
    (alert as any)?.created_at ||
    (alert as any)?.telemetry_time ||
    (alert as any)?.telemetry_timestamp ||
    "";
  const metric =
    alert.tag_name ||
    (alert as any)?.tag ||
    (alert as any)?.metric ||
    (alert as any)?.parameter ||
    alert.message ||
    "";
  const severity = alert.severity || "";

  if (did && (timestamp || metric)) {
    return `composite_${did}_${timestamp}_${metric}_${severity}`.trim();
  }

  // Do not deduplicate if no unique alert ID or metric can be determined
  return "";
}

export function getUniqueAlertId(alert: AlertItem | null | undefined): string {
  if (!alert) return "";
  const primary = alert._id || alert.alert_id || (alert as any)?.id;
  if (primary) return String(primary).trim();
  return getAlertDeduplicationKey(alert);
}
